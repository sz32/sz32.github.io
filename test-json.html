🎨 Sample Customizations

Add a new skill:

{
  "skills": {
    "categories": [
      {
        "name": "New Category",
        "items": ["Skill 1", "Skill 2"]
      }
    ]
  }
}

Add a new project:

{
  "projects": {
    "items": [
      {
        "name": "My New Project",
        "type": "regular",
        "description": "Project description",
        "responsibilities": ["Task 1", "Task 2"]
      }
    ]
  }
}

Update profile information:

{
  "profile": {
    "name": "Your Name",
    "title": "Your Title",
    "description": "Your bio..."
  }
}
